def fmt(x: Any, digits: int = 3, percentage: bool = False) -> str:
    if x is None or (isinstance(x, (float, np.floating)) and not np.isfinite(x)):
        return "—"
    if isinstance(x, (float, np.floating, int, np.integer)):
        return f"{x*100:.2f}%" if percentage else f"{x:.{digits}f}"
    return str(x).replace("|", "\\|").replace("\n", " ")


def table(headers: list[str], rows: list[list[Any]]) -> str:
    return ("| " + " | ".join(headers) + " |\n| " + " | ".join("---" for _ in headers)
            + " |\n" + "\n".join("| "+" | ".join(fmt(x) for x in row)+" |" for row in rows)+"\n")


def write(path: Path, text: str, *, replace_authorized_readme: bool = False) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    mode = "w" if replace_authorized_readme else "x"
    with path.open(mode, encoding="utf-8", newline="\n") as handle:
        handle.write(text)


def selected(summary: Any, name: str, family: str = "ALL", region: str = "motion",
             channel: str = "accepted") -> Any:
    return summary[(summary.method == name) & (summary.family == family)
                   & (summary.region == region) & (summary.channel == channel)].iloc[0]


def method_table(summary: Any) -> str:
    rows = []
    for name in METHODS:
        c = selected(summary, name, channel="candidate")
        a = selected(summary, name)
        rows.append([f"[{name}]({BASE}/{name}/README.md)", fmt(c.finite_rmse_m_s),
                     fmt(c.coverage, percentage=True), fmt(a.finite_rmse_m_s),
                     fmt(a.coverage, percentage=True), fmt(a.failure_fraction, percentage=True)])
    return table(["方法", "候选有限 RMSE m/s", "候选覆盖", "支持后有限 RMSE m/s",
                  "支持后覆盖", "支持后失败含缺失"], rows)
